# 0.6「管家骨架」Implementation Plan

> **状态：已完成（v0.6.0，2026-10-26）。** 下面的勾选框没有逐条打勾，以本行为准；实际实现以代码和 docs/项目细节.md 为准。

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:executing-plans. Steps use checkbox (`- [ ]`) syntax.

**Goal:** 让歇一会从「到点闹钟」升级成能主动开口的管家：
- 管家有五种来意，所有推送都走同一个推送出口：先过打扰预算（每天 8 条），再去重、看时机，最后显示到灵动岛；
- AI 通过 4 个新工具和管家打交道：`get_context`（看现状）、`push`（推送）、`remember` / `forget`（记下、删掉要管家知道的事）；
- 设置里给出「AI 巡视」任务说明，复制一下就能装好，巡视大约 45 分钟一次；
- App 内置两条规则：开工、收工。

**Architecture:**
- 纯逻辑都放在 `Shared/`：数据（`Butler.swift`）、推送关卡（`PushGate`）、工具（`ButlerTools`）。小程序可以当场告诉 AI「推了 / 被拦下了，原因是什么」。
- `push` 不直接显示，而是在 store 里写一条待推送记录。主 App 收到数据变化的通知后，过关卡、在灵动岛上显示，并把结果写回去。
- `get_context` 需要的现状（日程、在不在电脑前、今天休息情况）只有主 App 知道。小程序发一个「请给现状」的 Darwin 通知，主 App 收到后才写 `context.json`，小程序最多等 1.5 秒。**平时一次都不写盘**，符合性能要求。
- `ButlerCoordinator` 负责开工、收工这两条规则，以及从 store 里取待推送的记录。灵动岛和「忙的时候先等」的逻辑复用 `ReminderCoordinator` 里现成的。

**设计文档:** `docs/superpowers/specs/2026-10-24-butler-v2-design.md`（含 2026-10-26 已确认的三项）

**和设计文档的差异（实现时以这里为准，Task 10 回写设计文档）：**
1. 「少提醒这类」改成点一次就停，设计文档里写的是点两次；可以在设置里恢复。
2. 打扰预算、已屏蔽的来意、下班时间都存在 store 的 `butler.prefs` 里，没有放进 `AppSettings`。这样小程序也能当场用关卡判断。
3. 0.6 的 `push` 不支持「点了打开某个链接」，主按钮只是确认收到。

```swift
import Foundation

/// 管家为什么现在来找你（设计文档「五种来意」）
enum PushIntent: String, Codable, CaseIterable, Sendable {
    case due        // 到点了：你交代过的事
    case urgent     // 快来不及了
    case forgot     // 你可能忘了
    case pause      // 你该停一停了
    case info       // 值得让你知道

    var label: String {
        switch self {
        case .due: "到点了"
        case .urgent: "快来不及了"
        case .forgot: "可能忘了"
        case .pause: "该停一停"
        case .info: "告诉你一声"
        }
    }
}

/// 一次推送（来自 AI 的 push，或 App 内置规则）
struct PushRecord: Codable, Equatable, Sendable {
    enum Status: String, Codable, Sendable {
        case pending    // 等主 App 处理
        case shown      // 已显示，用户还没点
        case acted      // 点了主按钮
        case dismissed  // 点了「知道了」或自动收起
        case muted      // 点了「少提醒这类」
        case blocked    // 被关卡拦下（reason 里写原因）
    }

    var id: String
    var intent: PushIntent
    var title: String
    /// 为什么是现在
    var why: String
    /// 主按钮文字（如「去看看」），nil 时用「好的」
    var primaryAction: String?
    /// 同一件事的标识，一天只推一次
    var dedupeKey: String?
    /// 客户端名，或 "内置"
    var source: String
    var createdAt: Date
    var status: Status = .pending
    var reason: String?
    var handledAt: Date?
}

/// AI 让管家记下的事（承诺、截止、在吃的药、最近的身体状况……）
struct ButlerNote: Codable, Equatable, Sendable {
    var id: String
    var text: String
    /// 到这个时刻失效；nil = 一直有效
    var until: Date?
    var source: String
    var createdAt: Date
}

struct ButlerPrefs: Codable, Equatable, Sendable {
    /// 主动推送（不含「到点了」）每天最多几条
    var dailyBudget = 8
    /// 点过「少提醒这类」的来意
    var mutedIntents: [PushIntent] = []
    /// 下班时间（收工小结在这时推）
    var workEnd: ClockTime = ClockTime(hour: 18, minute: 30)!
}

/// store.json 里管家的部分
struct ButlerData: Codable, Equatable, Sendable {
    var pushes: [PushRecord] = []
    var notes: [ButlerNote] = []
    var prefs = ButlerPrefs()

    func activeNotes(now: Date) -> [ButlerNote] {
        notes.filter { ($0.until ?? .distantFuture) > now }
    }

    /// 只保留 14 天的推送记录、过期 7 天以上的记事
    mutating func prune(now: Date) {
        pushes.removeAll { now.timeIntervalSince($0.createdAt) > 14 * 86400 }
        notes.removeAll { note in note.until.map { now.timeIntervalSince($0) > 7 * 86400 } ?? false }
    }
}

/// 主 App 给 AI 看的现状（只在 AI 来要时生成）
struct ButlerContext: Codable, Equatable, Sendable {
    struct Event: Codable, Equatable, Sendable {
        var title: String
        var start: String
        var end: String
    }

    var generatedAt: Date
    /// 「在电脑前」「离开了」「开会中」「休息中」「暂停提醒中」
    var state: String
    /// 离开了多少分钟（不在离开状态时为 nil）
    var awayMinutes: Int?
    /// 这一段已经连续工作多少分钟
    var workingStretchMinutes: Int
    /// 今天第一次开始用电脑的时刻（HH:mm）
    var firstActiveToday: String?
    var breaksCompletedToday: Int
    var breaksSkippedToday: Int
    /// 今天剩下的日程（没开「按日历安排」时为 nil）
    var todayEvents: [Event]?
}
```

```swift
import Foundation

/// 推送出口的关卡：所有推送（AI 的、内置规则的）都先过这里，防止管家变烦
enum PushGate {
    enum Decision: Equatable, Sendable {
        case allow
        case block(String)
    }

    static let sameIntentGap: TimeInterval = 30 * 60

    /// 已经推出去的（显示过、用户处理过的），被拦下和还在排队的不算
    private static func delivered(_ p: PushRecord) -> Bool {
        [.shown, .acted, .dismissed, .muted].contains(p.status)
    }

    private static func deliveredToday(_ b: ButlerData, now: Date, calendar: Calendar) -> [PushRecord] {
        b.pushes.filter { delivered($0) && calendar.isDate($0.handledAt ?? $0.createdAt, inSameDayAs: now) }
    }

    static func remainingBudget(_ b: ButlerData, now: Date, calendar: Calendar) -> Int {
        max(0, b.prefs.dailyBudget - deliveredToday(b, now: now, calendar: calendar).filter { $0.intent != .due }.count)
    }

    static func check(_ push: PushRecord, butler b: ButlerData, now: Date, calendar: Calendar) -> Decision {
        // 「到点了」是用户自己交代的事，只看去重
        if push.intent != .due, b.prefs.mutedIntents.contains(push.intent) {
            return .block("你设置了少提醒「\(push.intent.label)」这类")
        }
        let today = deliveredToday(b, now: now, calendar: calendar).filter { $0.id != push.id }
        if let key = push.dedupeKey, today.contains(where: { $0.dedupeKey == key }) {
            return .block("同一件事今天已经说过了")
        }
        guard push.intent != .due else { return .allow }
        if today.contains(where: { $0.intent == push.intent && now.timeIntervalSince($0.handledAt ?? $0.createdAt) < sameIntentGap }) {
            return .block("30 分钟内已经提醒过「\(push.intent.label)」")
        }
        if remainingBudget(b, now: now, calendar: calendar) <= 0 {
            return .block("今天的打扰预算（\(b.prefs.dailyBudget) 条）用完了")
        }
        return .allow
    }
}
```

```swift
/// 管家的 4 个工具。和 ReminderTools 一样只依赖 Foundation；
/// situation 由调用方提供：小程序向主 App 要，测试里直接给
enum ButlerTools {

    - remember / forget：用户聊到承诺、截止、在吃的药、身体状况等值得管家记住的事时，征得同意后 remember，带上合适的 until。

    static func call(_ name: String, arguments args: [String: Any], context c: ToolContext,
                     situation: () -> ButlerContext?) -> ToolResult? {

    private static func getContext(_ c: ToolContext, situation: ButlerContext?) throws -> ToolResult {
        let data = try c.store.load()
        let upcoming = ReminderPlanner.plan(data, from: c.now, days: 1, limit: 10, calendar: c.calendar).compactMap { fire -> [String: Any]? in
            data.reminder(fire.reminderID).map { ["title": $0.title, "time": ScheduleText.fireText(fire.ringAt, now: c.now, calendar: c.calendar)] }
        }
        let todayPushes = data.butler.pushes.filter { c.calendar.isDate($0.createdAt, inSameDayAs: c.now) }.map {
            ["title": $0.title, "intent": $0.intent.label, "status": $0.status.rawValue] as [String: Any]
        }
        var object: [String: Any] = [
            "now": ["local": stamp(c.now, c), "weekday": "周" + ScheduleText.weekdayNames[ReminderPlanner.isoWeekday(c.now, calendar: c.calendar) - 1]],

                var item: [String: Any] = ["id": note.id, "text": note.text, "from": ClientNames.display(note.source)]
                if let until = note.until { item["until"] = ScheduleText.dateTime(until, calendar: c.calendar) }
                return item
            },
            "pushesToday": todayPushes,
            "remainingPushBudget": PushGate.remainingBudget(data.butler, now: c.now, calendar: c.calendar),
        ]
        if let s = situation, let json = try? JSONSerialization.jsonObject(with: ReminderStore.encoder.encode(s)) {
            object["situation"] = json
        } else {
            object["situation"] = "歇一会没在运行或没有回应，拿不到电脑前的状况"
        }
        return result(object, changed: false)

    private static func push(_ args: [String: Any], _ c: ToolContext) throws -> ToolResult {
        guard let raw = args["intent"] as? String, let intent = PushIntent(rawValue: raw) else {
            throw ToolError("intent 必须是 due / urgent / forgot / pause / info 之一")
        }
        let title = try text(args, "title", max: 20, required: true)!
        let why = try text(args, "why", max: 60, required: true)!
        let primary = try text(args, "primaryAction", max: 6, required: false)
        let key = try text(args, "dedupeKey", max: 60, required: false)
        return try c.store.update { data in
            var record = PushRecord(id: ReminderData.newID(existing: Set(data.butler.pushes.map(\.id))), intent: intent,
                                    title: title, why: why, primaryAction: primary, dedupeKey: key,
                                    source: c.client, createdAt: c.now)
            // 排队中的同一件事也算重复
            if let key, data.butler.pushes.contains(where: { $0.status == .pending && $0.dedupeKey == key }) {
                return result(["result": "没有推送：同一件事已经在排队了"], changed: false)
            }
            if case .block(let reason) = PushGate.check(record, butler: data.butler, now: c.now, calendar: c.calendar) {
                return result(["result": "没有推送：\(reason)"], changed: false)
            }
            record.status = .pending
            data.butler.pushes.append(record)
            data.butler.prune(now: c.now)
            return result(["result": "已交给歇一会，会在合适的时机显示",
                           "remainingPushBudget": PushGate.remainingBudget(data.butler, now: c.now, calendar: c.calendar)],
                          changed: true)
        }
    }

    private static func remember(_ args: [String: Any], _ c: ToolContext) throws -> ToolResult {
        let body = try text(args, "text", max: 100, required: true)!
        var until: Date?
        if let value = args["until"] as? String {
            let f = DateFormatter()
            f.calendar = c.calendar; f.timeZone = c.calendar.timeZone
            f.locale = Locale(identifier: "en_US_POSIX"); f.dateFormat = "yyyy-MM-dd"
            guard let day = f.date(from: value) else { throw ToolError("until 要写成 yyyy-MM-dd") }
            until = c.calendar.date(bySettingHour: 23, minute: 59, second: 59, of: day)
            guard until! > c.now else { throw ToolError("until 已经过去了") }
        }
        return try c.store.update { data in
            guard data.butler.notes.count < 100 else { throw ToolError("管家已经记了 100 件事，请先 forget 一些") }
            let note = ButlerNote(id: ReminderData.newID(existing: Set(data.butler.notes.map(\.id))), text: body,
                                  until: until, source: c.client, createdAt: c.now)
            data.butler.notes.append(note)
            return result(["remembered": ["id": note.id, "text": note.text]], changed: true)
        }
    }

    private static func forget(_ args: [String: Any], _ c: ToolContext) throws -> ToolResult {
        let id = try text(args, "id", max: 20, required: true)!
        return try c.store.update { data in
            guard let note = data.butler.notes.first(where: { $0.id == id }) else {
                throw ToolError("没有 id 为 \(id) 的记事，先用 get_context 看看")
            }
            data.butler.notes.removeAll { $0.id == id }
            return result(["forgot": note.text], changed: true)
        }
    }

    private static func text(_ args: [String: Any], _ key: String, max: Int, required: Bool) throws -> String? {
        guard let value = (args[key] as? String)?.trimmingCharacters(in: .whitespacesAndNewlines), !value.isEmpty else {
            if required { throw ToolError("缺少 \(key)") }
            return nil
        }
        guard value.count <= max else { throw ToolError("\(key) 最多 \(max) 个字") }
        return value
    }

    private static func stamp(_ date: Date, _ c: ToolContext) -> String {
        let f = DateFormatter()
        f.calendar = c.calendar; f.timeZone = c.calendar.timeZone
        f.locale = Locale(identifier: "en_US_POSIX"); f.dateFormat = "yyyy-MM-dd HH:mm"
        return f.string(from: date)
    }

    private static func result(_ object: [String: Any], changed: Bool) -> ToolResult {
        let data = (try? JSONSerialization.data(withJSONObject: object, options: [.prettyPrinted, .sortedKeys, .withoutEscapingSlashes])) ?? Data()
        return ToolResult(text: String(decoding: data, as: UTF8.self), isError: false, changed: changed)
    }
}
```

```swift
import AppKit
import Foundation

/// 内置规则（纯逻辑）：开工、收工。生成的推送和 AI 的一样过关卡
enum ButlerRules {
    static func isStartWindow(_ now: Date, calendar: Calendar) -> Bool {
        calendar.component(.hour, from: now) >= 6
    }

    static func dayKey(_ now: Date) -> String {
        let f = DateFormatter()
        f.locale = Locale(identifier: "en_US_POSIX"); f.dateFormat = "yyyy-MM-dd"
        return f.string(from: now)
    }

    /// 当天第一次开始用电脑：今天的安排。什么都没有就不推
    static func startOfDay(now: Date, events: [ButlerContext.Event], reminders: [(time: String, title: String)],
                           notes: [String]) -> PushRecord? {
        guard !(events.isEmpty && reminders.isEmpty && notes.isEmpty) else { return nil }
        var parts: [String] = []
        if !events.isEmpty { parts.append(events.prefix(3).map { "\($0.start) \($0.title)" }.joined(separator: "，")) }
        if !reminders.isEmpty { parts.append("提醒：" + reminders.prefix(3).map { "\($0.time) \($0.title)" }.joined(separator: "，")) }
        if let note = notes.first { parts.append("别忘了：\(note)") }
        let counts = [events.isEmpty ? nil : "\(events.count) 个日程", reminders.isEmpty ? nil : "\(reminders.count) 条提醒"].compactMap { $0 }
        return PushRecord(id: "start-\(dayKey(now))", intent: .info,
                          title: counts.isEmpty ? "今天要记着的事" : "今天有" + counts.joined(separator: "、"),
                          why: String(parts.joined(separator: "；").prefix(60)), primaryAction: "知道了",
                          dedupeKey: "start-\(dayKey(now))", source: "内置", createdAt: now)
    }

    /// 下班时间到了：今天剩下的事。没事就不推
    static func endOfDay(now: Date, remainingReminders: [(time: String, title: String)], missedToday: [String],
                         notes: [String]) -> PushRecord? {
        let count = remainingReminders.count + missedToday.count + notes.count
        guard count > 0 else { return nil }
        var parts: [String] = []
        if !missedToday.isEmpty { parts.append("错过了：" + missedToday.prefix(3).joined(separator: "、")) }
        if !remainingReminders.isEmpty { parts.append("今晚还有：" + remainingReminders.prefix(3).map { "\($0.time) \($0.title)" }.joined(separator: "，")) }
        if let note = notes.first { parts.append("别忘了：\(note)") }
        return PushRecord(id: "end-\(dayKey(now))", intent: .info, title: "收个尾：还有 \(count) 件事",
                          why: String(parts.joined(separator: "；").prefix(60)), primaryAction: "知道了",
                          dedupeKey: "end-\(dayKey(now))", source: "内置", createdAt: now)
    }
}

/// 管家：把待推送的记录过关卡、显示到灵动岛；跑开工、收工规则；AI 要现状时生成 context.json
@MainActor
final class ButlerCoordinator: NSObject {
    /// 由 BreakCoordinator 提供：生成现状、判断是否在电脑前、今天的日程
    var situation: () -> ButlerContext? = { nil }
    var isAtComputer: () -> Bool = { false }

    private let store: ReminderStore?
    private let reminders: ReminderCoordinator
    private var calendar: Calendar { Calendar.current }
    private var endTask: Task<Void, Never>?
    private static let startedKey = "butlerStartedDay"

    init(store: ReminderStore? = try? ReminderStore.shared(), reminders: ReminderCoordinator) {
        self.store = store
        self.reminders = reminders
        super.init()
    }

    func start() {
        NotificationCenter.default.addObserver(self, selector: #selector(storeChanged), name: .reminderStoreChanged, object: nil)
        CFNotificationCenterAddObserver(CFNotificationCenterGetDarwinNotifyCenter(), nil, { _, _, _, _, _ in
            DispatchQueue.main.async { NotificationCenter.default.post(name: .butlerContextRequested, object: nil) }
        }, DarwinNotice.contextRequest as CFString, nil, .deliverImmediately)
        NotificationCenter.default.addObserver(self, selector: #selector(writeContext), name: .butlerContextRequested, object: nil)
        processPending()
        scheduleEndOfDay()
    }

    @objc private func storeChanged() {
        processPending()
        scheduleEndOfDay()   // 下班时间可能改了
    }

    // MARK: - AI 要现状

    @objc private func writeContext() {
        guard let url = AppGroup.contextURL, let context = situation(),
              let data = try? ReminderStore.encoder.encode(context) else { return }
        try? data.write(to: url, options: .atomic)
    }

    // MARK: - 推送出口

    /// 取出 pending 的记录：过关卡，放行的显示到灵动岛，拦下的写原因
    func processPending(now: Date = Date()) {
        guard let store else { return }
        let toShow: [PushRecord] = (try? store.update(notify: false) { data in
            var shown: [PushRecord] = []
            for i in data.butler.pushes.indices where data.butler.pushes[i].status == .pending {
                let decision = PushGate.check(data.butler.pushes[i], butler: data.butler, now: now, calendar: calendar)
                data.butler.pushes[i].handledAt = now
                switch decision {
                case .allow:
                    data.butler.pushes[i].status = .shown
                    shown.append(data.butler.pushes[i])
                case .block(let reason):
                    data.butler.pushes[i].status = .blocked
                    data.butler.pushes[i].reason = reason
                }
            }
            return shown
        }) ?? []
        for push in toShow { reminders.showIsland(item(for: push)) }
    }

    /// 内置规则产生的推送：写进 store，再走同一个出口
    func enqueue(_ push: PushRecord) {
        guard let store else { return }
        _ = try? store.update(notify: false) { data in
            guard !data.butler.pushes.contains(where: { $0.id == push.id }) else { return }
            data.butler.pushes.append(push)
            data.butler.prune(now: Date())
        }
        processPending()
    }

    private func item(for push: PushRecord) -> IslandItem {
        let intent = IslandIntent(push.intent)
        var actions: [IslandItem.Action] = []
        if push.intent != .due {
            actions.append(.init(title: "少提醒这类") { [weak self] in self?.finish(push, .muted) })
        }
        actions.append(.init(title: "知道了") { [weak self] in self?.finish(push, .dismissed) })
        if let primary = push.primaryAction, primary != "知道了" {
            actions.append(.init(title: primary, primary: true) { [weak self] in self?.finish(push, .acted) })
        } else {
            actions[actions.count - 1] = .init(title: "知道了", primary: true) { [weak self] in self?.finish(push, .dismissed) }
        }
        return IslandItem(key: "push|\(push.id)", intent: intent, symbol: symbol(push.intent), kind: nil,
                          compact: String(push.title.prefix(4)), title: push.title, message: push.why,
                          trailing: TimeText.hourMinute(push.createdAt), firm: false, autoDismiss: 20, actions: actions,
                          onTimeout: { [weak self] in self?.record(push, .dismissed) })
    }

    private func symbol(_ intent: PushIntent) -> String {
        switch intent {
        case .due: "bell.fill"
        case .urgent: "clock.badge.exclamationmark.fill"
        case .forgot: "exclamationmark.bubble.fill"
        case .pause: "cup.and.saucer.fill"
        case .info: "sun.horizon.fill"
        }
    }

    private func finish(_ push: PushRecord, _ status: PushRecord.Status) {
        reminders.dismissIsland(key: "push|\(push.id)")
        record(push, status)
    }

    private func record(_ push: PushRecord, _ status: PushRecord.Status) {
        _ = try? store?.update(notify: false) { data in
            guard let i = data.butler.pushes.firstIndex(where: { $0.id == push.id }) else { return }
            data.butler.pushes[i].status = status
            data.butler.pushes[i].handledAt = Date()
            if status == .muted, !data.butler.prefs.mutedIntents.contains(push.intent) {
                data.butler.prefs.mutedIntents.append(push.intent)
            }
        }
    }

    // MARK: - 开工 / 收工

    /// BreakCoordinator 在「回到电脑前、开始工作」时调用（每次 tick 进入 running 时都调也没关系，一天只推一次）
    func userIsActive(now: Date = Date()) {
        let today = ButlerRules.dayKey(now)
        guard ButlerRules.isStartWindow(now, calendar: calendar),
              UserDefaults.standard.string(forKey: Self.startedKey) != today else { return }
        UserDefaults.standard.set(today, forKey: Self.startedKey)
        let context = situation()
        let data = (try? store?.load()) ?? ReminderData()
        let todayReminders = ReminderPlanner.plan(data, from: calendar.startOfDay(for: now), days: 1, limit: 20, calendar: calendar)
            .filter { calendar.isDate($0.ringAt, inSameDayAs: now) && $0.ringAt >= now }
            .compactMap { fire in data.reminder(fire.reminderID).map { (TimeText.hourMinute(fire.ringAt), $0.title) } }
        if let push = ButlerRules.startOfDay(now: now, events: context?.todayEvents ?? [], reminders: todayReminders,
                                             notes: data.butler.activeNotes(now: now).map(\.text)) {
            enqueue(push)
        }
    }

    private func scheduleEndOfDay() {
        endTask?.cancel()
        let data = (try? store?.load()) ?? ReminderData()
        let end = data.butler.prefs.workEnd
        let now = Date()
        guard var fire = calendar.date(bySettingHour: end.hour, minute: end.minute, second: 0, of: now) else { return }
        if fire <= now { fire = calendar.date(byAdding: .day, value: 1, to: fire) ?? fire }
        endTask = Task { [weak self] in
            try? await Task.sleep(for: .seconds(fire.timeIntervalSinceNow), tolerance: .seconds(30))
            guard let self, !Task.isCancelled else { return }
            self.endOfDay()
            self.scheduleEndOfDay()
        }
    }

    private func endOfDay(now: Date = Date()) {
        // 人不在电脑前就不推（推了也看不到）
        guard isAtComputer(), let store, let data = try? store.load() else { return }
        let remaining = ReminderPlanner.plan(data, from: now, days: 1, limit: 20, calendar: calendar)
            .filter { calendar.isDate($0.ringAt, inSameDayAs: now) }
            .compactMap { fire in data.reminder(fire.reminderID).map { (TimeText.hourMinute(fire.ringAt), $0.title) } }
        let missed = data.occurrences.filter { $0.outcome == .missed && calendar.isDate($0.scheduledAt, inSameDayAs: now) }
            .compactMap { data.reminder($0.reminderID)?.title }
        let notesToday = data.butler.activeNotes(now: now).filter { note in
            note.until.map { calendar.isDate($0, inSameDayAs: now) } ?? false
        }.map(\.text)
        if let push = ButlerRules.endOfDay(now: now, remainingReminders: remaining, missedToday: missed, notes: notesToday) {
            enqueue(push)
        }
    }
}
```
